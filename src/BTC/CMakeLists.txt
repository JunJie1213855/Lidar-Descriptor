cmake_minimum_required(VERSION 3.8)
project(btc_desc)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters kdtree)
find_package(OpenCV REQUIRED)
find_package(Ceres REQUIRED)
find_package(TBB REQUIRED)

add_library(btc_descriptor_core src/btc.cpp)
target_include_directories(btc_descriptor_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
  ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(btc_descriptor_core rclcpp sensor_msgs visualization_msgs geometry_msgs pcl_conversions cv_bridge)
target_link_libraries(btc_descriptor_core ${PCL_LIBRARIES} ${OpenCV_LIBRARIES} Ceres::ceres TBB::tbb)
add_executable(btc_extract_node ros2/btc_extract_node.cpp)
ament_target_dependencies(btc_extract_node rclcpp sensor_msgs pcl_conversions)
target_link_libraries(btc_extract_node btc_descriptor_core ${PCL_LIBRARIES})
install(TARGETS btc_extract_node DESTINATION lib/${PROJECT_NAME})
install(TARGETS btc_descriptor_core EXPORT export_btc_descriptor_core
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(FILES ros2/launch/btc_extract.launch.py DESTINATION share/${PROJECT_NAME}/launch)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_btc_descriptor_core HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp sensor_msgs visualization_msgs geometry_msgs pcl_conversions cv_bridge PCL OpenCV Ceres TBB)
ament_package()
